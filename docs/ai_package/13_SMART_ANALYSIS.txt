import 'dart:io';
import 'dart:math' as math;
import 'package:image/image.dart' as img;

class AnalysisSuggestion {
  final String id;
  final String title;
  final String description;
  final String iconName;
  final String action; // 'removeBg', 'enhance', 'shadow', 'crop', 'export'
  final int priority;

  const AnalysisSuggestion({
    required this.id,
    required this.title,
    required this.description,
    required this.iconName,
    required this.action,
    required this.priority,
  });
}

class AnalysisResult {
  final bool ok;
  final String? error;
  final List<AnalysisSuggestion> suggestions;
  final Map<String, dynamic> metadata;

  const AnalysisResult({
    required this.ok,
    this.error,
    required this.suggestions,
    this.metadata = const {},
  });
}

class SmartAnalysisService {
  Future<AnalysisResult> analyze(String imagePath) async {
    try {
      final bytes = await File(imagePath).readAsBytes();
      final im = img.decodeImage(bytes);
      if (im == null) {
        return const AnalysisResult(ok: false, error: 'Decode failed', suggestions: []);
      }

      final suggestions = <AnalysisSuggestion>[];

      // ─── 1. Brightness ───
      final brightness = _avgBrightness(im);
      if (brightness < 90) {
        suggestions.add(const AnalysisSuggestion(
          id: 'dark_image',
          title: 'Image is dark',
          description: 'Brightness low. Enhance lighting?',
          iconName: 'brightness_6',
          action: 'enhance',
          priority: 90,
        ));
      } else if (brightness > 220) {
        suggestions.add(const AnalysisSuggestion(
          id: 'overexposed',
          title: 'Image is overexposed',
          description: 'Too bright. Reduce highlights?',
          iconName: 'brightness_4',
          action: 'enhance',
          priority: 80,
        ));
      }

      // ─── 2. Resolution ───
      if (im.width < 1000 || im.height < 1000) {
        suggestions.add(AnalysisSuggestion(
          id: 'low_resolution',
          title: 'Low resolution',
          description: '${im.width}×${im.height}. Upscale 2×?',
          iconName: 'high_quality',
          action: 'enhance',
          priority: 95,
        ));
      }

      // ─── 3. Background complexity ───
      final bgComplexity = _bgComplexity(im);
      if (bgComplexity > 0.3) {
        suggestions.add(const AnalysisSuggestion(
          id: 'complex_bg',
          title: 'Complex background',
          description: 'Remove background for cleaner look?',
          iconName: 'content_cut',
          action: 'removeBg',
          priority: 100,
        ));
      } else if (bgComplexity < 0.05) {
        suggestions.add(const AnalysisSuggestion(
          id: 'solid_bg',
          title: 'Solid background',
          description: 'Fast background removal available',
          iconName: 'auto_fix_high',
          action: 'removeBg',
          priority: 70,
        ));
      }

      // ─── 4. Shadow ───
      if (!_hasShadow(im) && bgComplexity < 0.3) {
        suggestions.add(const AnalysisSuggestion(
          id: 'no_shadow',
          title: 'No shadow detected',
          description: 'Add realistic shadow for depth?',
          iconName: 'brightness_4',
          action: 'shadow',
          priority: 60,
        ));
      }

      // ─── 5. Coverage ───
      final coverage = _coverage(im);
      if (coverage < 0.4) {
        suggestions.add(AnalysisSuggestion(
          id: 'small_product',
          title: 'Product appears small',
          description: 'Product fills ${(coverage * 100).toStringAsFixed(0)}%. Crop closer?',
          iconName: 'crop',
          action: 'crop',
          priority: 65,
        ));
      }

      // ─── 6. Aspect ratio ───
      final ratio = im.width / im.height;
      if (ratio < 0.9 || ratio > 1.1) {
        suggestions.add(const AnalysisSuggestion(
          id: 'aspect_ratio',
          title: 'Non-square aspect',
          description: 'Square (1:1) preferred for marketplaces',
          iconName: 'aspect_ratio',
          action: 'crop',
          priority: 55,
        ));
      }

      suggestions.sort((a, b) => b.priority.compareTo(a.priority));

      return AnalysisResult(
        ok: true,
        suggestions: suggestions.take(5).toList(),
        metadata: {
          'width': im.width,
          'height': im.height,
          'brightness': brightness.round(),
          'bg_complexity': bgComplexity,
          'coverage': coverage,
        },
      );
    } catch (e) {
      return AnalysisResult(ok: false, error: '$e', suggestions: []);
    }
  }

  double _avgBrightness(img.Image im) {
    var sum = 0.0, count = 0;
    final step = math.max(1, math.min(im.width, im.height) ~/ 100);
    for (var y = 0; y < im.height; y += step) {
      for (var x = 0; x < im.width; x += step) {
        final p = im.getPixel(x, y);
        sum += 0.299 * p.r + 0.587 * p.g + 0.114 * p.b;
        count++;
      }
    }
    return count == 0 ? 0 : sum / count;
  }

  double _bgComplexity(img.Image im) {
    final samples = <List<int>>[];
    final step = math.max(1, math.min(im.width, im.height) ~/ 50);
    for (var x = 0; x < im.width; x += step) {
      samples.add([im.getPixel(x, 5).r.toInt(), im.getPixel(x, 5).g.toInt(), im.getPixel(x, 5).b.toInt()]);
      samples.add([im.getPixel(x, im.height - 6).r.toInt(), im.getPixel(x, im.height - 6).g.toInt(), im.getPixel(x, im.height - 6).b.toInt()]);
    }
    for (var y = 0; y < im.height; y += step) {
      samples.add([im.getPixel(5, y).r.toInt(), im.getPixel(5, y).g.toInt(), im.getPixel(5, y).b.toInt()]);
      samples.add([im.getPixel(im.width - 6, y).r.toInt(), im.getPixel(im.width - 6, y).g.toInt(), im.getPixel(im.width - 6, y).b.toInt()]);
    }
    if (samples.isEmpty) return 0;
    var r = 0.0, g = 0.0, b = 0.0;
    for (final s in samples) { r += s[0]; g += s[1]; b += s[2]; }
    r /= samples.length; g /= samples.length; b /= samples.length;
    var v = 0.0;
    for (final s in samples) {
      v += math.pow(s[0] - r, 2) + math.pow(s[1] - g, 2) + math.pow(s[2] - b, 2);
    }
    v /= samples.length * 3;
    return math.sqrt(v) / 128.0;
  }

  bool _hasShadow(img.Image im) {
    final startY = (im.height * 2) ~/ 3;
    var darkCount = 0, total = 0;
    final step = math.max(1, im.height ~/ 100);
    for (var y = startY; y < im.height; y += step) {
      for (var x = 0; x < im.width; x += step) {
        final p = im.getPixel(x, y);
        final b = 0.299 * p.r + 0.587 * p.g + 0.114 * p.b;
        if (b < 100) darkCount++;
        total++;
      }
    }
    return total > 0 && (darkCount / total) > 0.15;
  }

  double _coverage(img.Image im) {
    final borderColor = [
      im.getPixel(5, 5).r.toInt(),
      im.getPixel(5, 5).g.toInt(),
      im.getPixel(5, 5).b.toInt(),
    ];
    var productPixels = 0, totalPixels = 0;
    final step = math.max(1, im.width ~/ 50);
    for (var y = 0; y < im.height; y += step) {
      for (var x = 0; x < im.width; x += step) {
        final p = im.getPixel(x, y);
        final dist = math.sqrt(
          math.pow(p.r - borderColor[0], 2) +
          math.pow(p.g - borderColor[1], 2) +
          math.pow(p.b - borderColor[2], 2),
        );
        if (dist > 60) productPixels++;
        totalPixels++;
      }
    }
    return totalPixels > 0 ? productPixels / totalPixels : 0.0;
  }
}

import 'package:flutter/material.dart';
import 'package:flutter_animate/flutter_animate.dart';
import '../core/theme.dart';
import '../services/smart_analysis_service.dart';

class SmartAnalysisPanel extends StatelessWidget {
  final AnalysisResult result;
  final ValueChanged<String> onSuggestionTap;
  final VoidCallback onDismiss;

  const SmartAnalysisPanel({
    super.key,
    required this.result,
    required this.onSuggestionTap,
    required this.onDismiss,
  });

  @override
  Widget build(BuildContext context) {
    if (!result.ok || result.suggestions.isEmpty) return const SizedBox.shrink();

    return Container(
      margin: const EdgeInsets.all(16),
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        gradient: LinearGradient(
          colors: [
            AppColors.primary.withValues(alpha: 0.15),
            AppColors.primaryGlow.withValues(alpha: 0.08),
          ],
        ),
        borderRadius: BorderRadius.circular(16),
        border: Border.all(
          color: AppColors.primary.withValues(alpha: 0.3),
          width: 1.5,
        ),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Row(
            children: [
              const Icon(Icons.auto_awesome, color: AppColors.primary, size: 20),
              const SizedBox(width: 8),
              const Text('Smart Analysis',
                  style: TextStyle(
                    color: AppColors.textPrimary,
                    fontSize: 14,
                    fontWeight: FontWeight.w700,
                  )),
              const Spacer(),
              IconButton(
                icon: const Icon(Icons.close, size: 18, color: AppColors.textTertiary),
                onPressed: onDismiss,
                padding: EdgeInsets.zero,
                constraints: const BoxConstraints(),
              ),
            ],
          ),
          const SizedBox(height: 12),
          ...result.suggestions.take(3).map((s) => Padding(
                padding: const EdgeInsets.only(bottom: 8),
                child: InkWell(
                  onTap: () => onSuggestionTap(s.action),
                  borderRadius: BorderRadius.circular(12),
                  child: Container(
                    padding: const EdgeInsets.all(12),
                    decoration: BoxDecoration(
                      color: AppColors.surface,
                      borderRadius: BorderRadius.circular(12),
                      border: Border.all(color: AppColors.border),
                    ),
                    child: Row(
                      children: [
                        Container(
                          width: 36,
                          height: 36,
                          decoration: BoxDecoration(
                            color: AppColors.primary.withValues(alpha: 0.15),
                            borderRadius: BorderRadius.circular(10),
                          ),
                          child: Icon(_iconFor(s.iconName),
                              color: AppColors.primary, size: 18),
                        ),
                        const SizedBox(width: 12),
                        Expanded(
                          child: Column(
                            crossAxisAlignment: CrossAxisAlignment.start,
                            children: [
                              Text(s.title,
                                  style: const TextStyle(
                                    color: AppColors.textPrimary,
                                    fontSize: 13,
                                    fontWeight: FontWeight.w600,
                                  )),
                              const SizedBox(height: 2),
                              Text(s.description,
                                  style: const TextStyle(
                                    color: AppColors.textSecondary,
                                    fontSize: 11,
                                  ),
                                  maxLines: 2,
                                  overflow: TextOverflow.ellipsis),
                            ],
                          ),
                        ),
                        const Icon(Icons.arrow_forward_ios,
                            size: 14, color: AppColors.textTertiary),
                      ],
                    ),
                  ),
                ),
              )),
        ],
      ),
    ).animate().fadeIn().slideY(begin: -0.1, duration: 400.ms);
  }

  IconData _iconFor(String name) => switch (name) {
        'brightness_6' => Icons.brightness_6,
        'brightness_4' => Icons.brightness_4,
        'high_quality' => Icons.high_quality,
        'content_cut' => Icons.content_cut,
        'auto_fix_high' => Icons.auto_fix_high,
        'crop' => Icons.crop,
        'aspect_ratio' => Icons.aspect_ratio,
        _ => Icons.auto_awesome,
      };
}

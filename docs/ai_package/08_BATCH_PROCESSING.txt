class BatchJob {
  final String inputPath;
  final String outputPath;
  final bool ok;
  final String? error;
  final Duration duration;

  const BatchJob({
    required this.inputPath,
    required this.outputPath,
    required this.ok,
    this.error,
    this.duration = Duration.zero,
  });
}

class BatchProgress {
  final int total;
  final int completed;
  final int failed;
  final String? currentStep;
  final bool running;
  final bool cancelled;

  const BatchProgress({
    this.total = 0,
    this.completed = 0,
    this.failed = 0,
    this.currentStep,
    this.running = false,
    this.cancelled = false,
  });

  double get fraction => total == 0 ? 0.0 : (completed + failed) / total;
  bool get isDone => (completed + failed) >= total && total > 0;
  int get remaining => total - completed - failed;

  BatchProgress copyWith({
    int? total,
    int? completed,
    int? failed,
    String? currentStep,
    bool? running,
    bool? cancelled,
  }) =>
      BatchProgress(
        total: total ?? this.total,
        completed: completed ?? this.completed,
        failed: failed ?? this.failed,
        currentStep: currentStep ?? this.currentStep,
        running: running ?? this.running,
        cancelled: cancelled ?? this.cancelled,
      );
}

class BatchService extends StateNotifier<BatchProgress> {
  BatchService() : super(const BatchProgress());

  final _border = BorderCutService();
  final _shadow = ShadowService();
  final _seika = SeikaService(quality: 'best');
  final _export = ExportService();

  bool _cancelRequested = false;

  Future<List<BatchJob>> processAll(
    List<String> inputs, {
    bool addShadow = true,
    String format = 'png',
    int? size = 2048,
    String? backgroundPreset,
    Function(BatchProgress)? onProgress,
  }) async {
    _cancelRequested = false;
    state = BatchProgress(total: inputs.length, running: true);
    final results = <BatchJob>[];
    var completed = 0;
    var failed = 0;

    for (var i = 0; i < inputs.length; i++) {
      if (_cancelRequested) {
        state = state.copyWith(
          running: false,
          cancelled: true,
          currentStep: 'Cancelled at ${i + 1}/${inputs.length}',
        );
        break;
      }

      final input = inputs[i];
      final sw = Stopwatch()..start();

      try {
        state = state.copyWith(
          completed: completed,
          failed: failed,
          currentStep: 'Removing BG (${i + 1}/${inputs.length})',
        );
        onProgress?.call(state);

        // ─── Step 1: Remove Background ───
        final r1 = await _border.removeBg(input);
        if (!r1.ok) throw Exception(r1.error ?? 'BG removal failed');

        String currentPath = r1.outputPath!;

        // ─── Step 2: Add Shadow (optional) ───
        if (addShadow) {
          state = state.copyWith(currentStep: 'Adding shadow (${i + 1}/${inputs.length})');
          onProgress?.call(state);
          final r2 = await _shadow.addShadow(currentPath);
          if (r2.ok && r2.outputPath != null) currentPath = r2.outputPath!;
        }

        // ─── Step 3: Export ───
        state = state.copyWith(currentStep: 'Exporting (${i + 1}/${inputs.length})');
        onProgress?.call(state);
        final file = await _export.export(
          currentPath,
          format: format,
          size: size,
        );

        results.add(BatchJob(
          inputPath: input,
          outputPath: file.path,
          ok: true,
          duration: sw.elapsed,
        ));
        completed++;
      } catch (e) {
        results.add(BatchJob(
          inputPath: input,
          outputPath: '',
          ok: false,
          error: '$e',
          duration: sw.elapsed,
        ));
        failed++;
      }
    }

    state = BatchProgress(
      total: inputs.length,
      completed: completed,
      failed: failed,
      running: false,
      cancelled: _cancelRequested,
      currentStep: _cancelRequested
          ? 'Cancelled'
          : 'Done: $completed ok, $failed failed',
    );
    onProgress?.call(state);

    return results;
  }

  void cancel() {
    _cancelRequested = true;
  }

  void reset() {
    state = const BatchProgress();
  }
}

final batchProvider = StateNotifierProvider<BatchService, BatchProgress>(
  (_) => BatchService(),
);

class _BatchScreenState extends ConsumerState<BatchScreen> {
  final _inputs = <String>[];
  final _results = <BatchJob>[];
  bool _addShadow = true;
  String _format = 'png';
  int _size = 2048;

  @override
  Widget build(BuildContext context) {
    final billing = ref.watch(billingProvider);
    final progress = ref.watch(batchProvider);
    final isPro = billing.proService.isPro;

    if (!FeatureFlags.batchProcessing) {
      return Scaffold(
        appBar: AppBar(title: const Text('Batch Processing')),
        body: const EmptyState(
          icon: Icons.construction,
          title: 'Coming Soon',
          subtitle: 'Batch processing is under development',
        ),
      );
    }

    if (!isPro) {
      return Scaffold(
        appBar: AppBar(title: const Text('Batch Processing')),
        body: _ProGate(onUpgrade: () => context.push('/paywall')),
      );
    }

    return Scaffold(
      appBar: AppBar(
        title: const Text('Batch Processing'),
        actions: [
          if (_results.isNotEmpty && !progress.running)
            IconButton(
              icon: const Icon(Icons.delete_sweep),
              onPressed: () => setState(() => _results.clear()),
              tooltip: 'Clear results',
            ),
          if (progress.running)
            IconButton(
              icon: const Icon(Icons.cancel),
              color: AppColors.danger,
              onPressed: () => ref.read(batchProvider.notifier).cancel(),
              tooltip: 'Cancel',
            ),
        ],
      ),
      body: Column(
        children: [
          _Toolbar(
            count: _inputs.length,
            running: progress.running,
            onPick: _pickImages,
            onProcess: _processAll,
          ),
          _Settings(
            addShadow: _addShadow,
            format: _format,
            size: _size,
            onShadowChange: (v) => setState(() => _addShadow = v),
            onFormatChange: (v) => setState(() => _format = v),
            onSizeChange: (v) => setState(() => _size = v),
          ),
          if (progress.running || progress.total > 0)
            _ProgressSection(progress: progress),
          Expanded(
            child: _inputs.isEmpty
                ? const EmptyState(
                    icon: Icons.photo_library,
                    title: 'No images selected',
                    subtitle: 'Select up to 100 images',
                  )
                : _Grid(
                    inputs: _inputs,
                    results: _results,
                    running: progress.running,
                  ),
          ),
          if (progress.isDone && _results.any((r) => r.ok))
            _ShareAll(results: _results, onShare: _shareAll),
        ],
      ),
    );
  }

  Future<void> _pickImages() async {
    final res = await FilePicker.platform.pickFiles(
      type: FileType.image,
      allowMultiple: true,
      withData: false,
    );
    if (res == null) return;

    final paths = res.paths.whereType<String>().toList();
    if (paths.length > AppConstants.maxBatchImages) {
      if (!mounted) return;
      showErrorSnack(context, 'Max ${AppConstants.maxBatchImages} images');
      return;
    }

    setState(() {
      _inputs
        ..clear()
        ..addAll(paths);
      _results.clear();
    });
  }

  Future<void> _processAll() async {
    final results = await ref.read(batchProvider.notifier).processAll(
          _inputs,
          addShadow: _addShadow,
          format: _format,
          size: _size,
        );
    if (!mounted) return;
    setState(() {
      _results
        ..clear()
        ..addAll(results);
    });
  }

  Future<void> _shareAll() async {
    final files = _results
        .where((r) => r.ok)
        .map((r) => XFile(r.outputPath))
        .toList();
    if (files.isEmpty) return;
    await Share.shareXFiles(files, text: 'Processed with ProductChat Studio');
  }
}

class _Grid extends StatelessWidget {
  final List<String> inputs;
  final List<BatchJob> results;
  final bool running;

  const _Grid({
    required this.inputs,
    required this.results,
    required this.running,
  });

  @override
  Widget build(BuildContext context) => GridView.builder(
        padding: const EdgeInsets.all(16),
        gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
          crossAxisCount: 3,
          crossAxisSpacing: 8,
          mainAxisSpacing: 8,
        ),
        itemCount: inputs.length,
        itemBuilder: (_, i) {
          final input = inputs[i];
          final result = results.firstWhere(
            (r) => r.inputPath == input,
            orElse: () => const BatchJob(
              inputPath: '',
              outputPath: '',
              ok: false,
            ),
          );
          final processed = result.inputPath.isNotEmpty;
          return _ImageTile(
            input: input,
            result: processed ? result : null,
            running: running,
          );
        },
      );
}
